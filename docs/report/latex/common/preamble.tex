\newbox\seq@box

\def\seq@placelabel#1{%
  \pgfinterruptpicture
    \global\setbox\seq@box\hbox{\sffamily\footnotesize\begin{tabular}{@{}c@{}}\the\seq@no:~#1\end{tabular}}%
  \endpgfinterruptpicture
  \seq@len=\seq@xb \advance\seq@len by -\seq@xa
  \ifdim\seq@len<0pt \seq@len=-\seq@len \fi
  \advance\seq@len by -0.25cm
  \ifdim\wd\seq@box<\seq@len
    \node[vplbl, above=0.5pt] at ($(\seq@P)!0.5!(\seq@Q)$) {\the\seq@no:~#1};
  \else
    \seq@len=\wd\seq@box
    \ifdim\seq@xa<\seq@xb
      \advance\seq@len by \seq@xa
      \ifdim\seq@len>\seq@maxx
        \node[vplbl, anchor=south east] at ($(\seq@Q)+(-0.06,0.02)$) {\the\seq@no:~#1};
      \else
        \node[vplbl, anchor=south west] at ($(\seq@P)+(0.06,0.02)$) {\the\seq@no:~#1};
      \fi
    \else
      \seq@len=-\seq@len \advance\seq@len by \seq@xa
      \ifdim\seq@len<\seq@minx
        \node[vplbl, anchor=south west] at ($(\seq@Q)+(0.06,0.02)$) {\the\seq@no:~#1};
      \else
        \node[vplbl, anchor=south east] at ($(\seq@P)+(-0.06,0.02)$) {\the\seq@no:~#1};
      \fi
    \fi
  \fi}
